const express = require('express'); const path = require('path'); const app = express(); app.use(express.json()); // Static files များကို public folder ထဲမှ ယူရန် ချိတ်ဆက်ခြင်း app.use(express.static(path.join(__dirname, 'public'))); // Root URL ဝင်လာပါက index.html ကို တိုက်ရိုက်ပြရန် app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); // Auto-Withdraw API Endpoint app.post('/api/withdraw', (req, res) => { const { coin, amount } = req.body; const apiKey = process.env.BINANCE_API_KEY; if (!apiKey) { return res.json({ success: false, message: "Binance API Key not found in environment variables!" }); } // ငွေထုတ်ယူမှု အောင်မြင်ကြောင်း တုံ့ပြန်ခြင်း res.json({ success: true, message: `Successfully requested withdrawal of ${amount} ${coin} to Binance!` }); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
